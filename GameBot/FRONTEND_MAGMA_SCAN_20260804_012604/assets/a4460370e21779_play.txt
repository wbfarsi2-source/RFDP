  <script>
    /* Tiny boot helper exposed before game.js runs. game.js / auth-gate.js call
       these as the boot lifecycle progresses, then hide the overlay after the
       first frame once auth/session is ready. Defined inline so it's available
       the instant the HTML is parsed, before the (much larger) module script
       starts.

       The overlay has two visual modes:
         - "connecting"  (default): logo + shard loader + phase label + progress bar.
         - "landing"     (auth gate visible): logo + gradient bg only, so the
                         wallet-connect UI on top reads as a clean landing page
                         without loader/progress noise during user choice.
       `showLanding()` and `showConnecting(label)` flip between the two. */
    (function () {
      /* The static hint advertises the scroll wheel — a control coarse-pointer
       * devices don't have (they pinch instead, see main-renderer.js). Swap the
       * copy before first paint settles so phones never see the desktop tip.
       * pointer:coarse, not maxTouchPoints: a touchscreen laptop still has a
       * wheel, so the desktop tip stays correct there. */
      try {
        if (window.matchMedia && window.matchMedia('(pointer: coarse)').matches) {
          var hintSwapEl = document.getElementById('kintara-boot-hint');
          if (hintSwapEl) hintSwapEl.textContent = 'Tip: pinch to zoom in & out';
        }
      } catch (_) { /* ignore */ }
      var phaseEl = document.getElementById('kintara-boot-phase');
      var barEl = document.getElementById('kintara-boot-bar');
      var overlayEl = document.getElementById('kintara-boot-overlay');
      var spinnerEl = overlayEl ? overlayEl.querySelector('.kintara-boot__spinner') : null;
      var barWrapEl = overlayEl ? overlayEl.querySelector('.kintara-boot__bar') : null;
      var hintEl = overlayEl ? overlayEl.querySelector('.kintara-boot__hint') : null;
      var subEl = overlayEl ? overlayEl.querySelector('.kintara-boot__sub') : null;
      var lastPct = 0;
      window.__kintaraLoadingPhase = function (label, pct) {
        try {
          if (phaseEl && typeof label === 'string' && label) phaseEl.textContent = label;
          if (typeof pct === 'number' && isFinite(pct)) {
            var p = Math.max(lastPct, Math.min(100, Math.round(pct)));
            lastPct = p;
            if (barEl) barEl.style.width = p + '%';
          }
        } catch (_) { /* ignore */ }
      };
      window.__kintaraLoadingShowLanding = function () {
        try {
          if (spinnerEl) spinnerEl.style.display = 'none';
          if (phaseEl) phaseEl.style.visibility = 'hidden';
          if (barWrapEl) barWrapEl.style.visibility = 'hidden';
          if (hintEl) hintEl.style.visibility = 'hidden';
          /* Also hide the "— Please wait —" sub: in landing mode the boot
             frame hosts a "Play" CTA (logged-out /play visitor), so any
             "please wait" copy contradicts what the user is being asked to
             do. `showConnecting` restores it for normal loading phases. */
          if (subEl) subEl.style.visibility = 'hidden';
        } catch (_) { /* ignore */ }
      };
      window.__kintaraLoadingShowConnecting = function (label) {
        try {
          var rb = document.getElementById('kintara-boot-retry');
          if (rb) rb.style.display = 'none';
          if (spinnerEl) spinnerEl.style.display = '';
          if (phaseEl) phaseEl.style.visibility = '';
          if (barWrapEl) barWrapEl.style.visibility = '';
          if (hintEl) hintEl.style.visibility = '';
          if (subEl) subEl.style.visibility = '';
          if (phaseEl && typeof label === 'string' && label) phaseEl.textContent = label;
        } catch (_) { /* ignore */ }
      };
      window.__kintaraLoadingHide = function () {
        try {
          if (!overlayEl) return;
          overlayEl.classList.add('kintara-boot--fading');
          setTimeout(function () {
            /* Keep the overlay in the DOM (just hidden) so it can be re-shown
               for in-game full-screen transitions like stalking a friend onto
               another shard. Removing it would force us to rebuild the markup. */
            try { if (overlayEl) overlayEl.style.display = 'none'; } catch (_) {}
          }, 600);
        } catch (_) { /* ignore */ }
      };
      /* Failure state — the overlay used to spin on "Please wait" forever
         when a connect path gave up (e.g. the spectate WS after its retry
         budget). Shows the message plus a Retry button that reloads. */
      window.__kintaraLoadingShowError = function (label) {
        try {
          if (!overlayEl) return;
          overlayEl.style.display = '';
          overlayEl.classList.remove('kintara-boot--fading');
          if (spinnerEl) spinnerEl.style.display = 'none';
          if (barWrapEl) barWrapEl.style.visibility = 'hidden';
          if (hintEl) hintEl.style.visibility = 'hidden';
          if (subEl) subEl.style.visibility = 'hidden';
          if (phaseEl) {
            phaseEl.style.visibility = '';
            phaseEl.textContent = typeof label === 'string' && label ? label : 'Connection failed.';
          }
          var btn = document.getElementById('kintara-boot-retry');
          if (!btn) {
            btn = document.createElement('button');
            btn.id = 'kintara-boot-retry';
            btn.type = 'button';
            btn.textContent = 'Retry';
            btn.style.cssText = 'margin-top:16px;padding:12px 34px;min-height:44px;border-radius:10px;border:2px solid rgba(100,140,180,0.5);background:linear-gradient(180deg,#fff0c8 0%,#ffb24a 35%,#d66a18 72%,#8f3d0c 100%);color:#3a1c04;font:800 14px/1 system-ui,"Segoe UI",Roboto,sans-serif;letter-spacing:0.08em;text-transform:uppercase;cursor:pointer;touch-action:manipulation;';
            btn.addEventListener('click', function () { try { location.reload(); } catch (_) { /* ignore */ } });
            var frame = overlayEl.querySelector('.kintara-boot__frame');
            if (frame) frame.appendChild(btn);
          }
          btn.style.display = '';
        } catch (_) { /* ignore */ }
      };
      /* Re-show the boot/loading overlay for an in-game server switch. Resets
         the progress bar so it animates fresh, restores connecting mode, and
         sets the initial phase label. */
      window.__kintaraLoadingShow = function (label, pct) {
        try {
          if (!overlayEl) return;
          lastPct = 0;
          if (barEl) barEl.style.width = (typeof pct === 'number' && isFinite(pct) ? Math.max(0, Math.min(100, Math.round(pct))) : 8) + '%';
          overlayEl.style.display = '';
          overlayEl.classList.remove('kintara-boot--fading');
          if (typeof window.__kintaraLoadingShowConnecting === 'function') {
            window.__kintaraLoadingShowConnecting(typeof label === 'string' ? label : undefined);
          } else if (phaseEl && typeof label === 'string' && label) {
            phaseEl.textContent = label;
          }
        } catch (_) { /* ignore */ }
      };
    })();

  <!-- Service Worker registration: static-asset cache only. Game JS/HTML stay
       network-fetched so updates apply immediately without a hard reload.
       Registered after the page paints so it never delays first contentful paint. -->
  <script>
    (function () {
      if (!('serviceWorker' in navigator)) return;
      var register = function () {
        navigator.serviceWorker
          .register('/sw.js', { scope: '/' })
          .catch(function () { /* registration failure is non-fatal */ });
        /* Deferred maintenance-page warm: the SW only pre-caches the critical
           maintenance set at install; the remaining slideshow backgrounds
           (~6 MB) are fetched here, a minute after load, so they never compete
           with realm assets on a first visit. The SW no-ops if already cached. */
        setTimeout(function () {
          navigator.serviceWorker.ready.then(function (reg) {
            if (reg && reg.active) reg.active.postMessage({ type: 'kintara-maint-prefetch-rest' });
          }).catch(function () { /* no SW — nothing to warm */ });
        }, 60000);
      };
      if (document.readyState === 'complete') register();
      else window.addEventListener('load', register);
    })();
  </script>
